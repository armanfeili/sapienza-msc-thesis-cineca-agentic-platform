🚀 QUICK DEPLOYMENT (5 Minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1. git clone <repo> && cd Cineca-Agentic-Platform
  2. cp .env.example .env
  3. # Edit .env: JWT_SECRET, DB_PASSWORD, REDIS_PASSWORD
  4. docker compose up -d
  5. docker compose ps  # Wait for healthy status
  6. curl http://localhost:8000/v1/health/ready
  7. Configure defaults via UI: http://localhost:8501
